cmake_minimum_required(VERSION 3.20)
# This standalone sample links an imported SDK. No private source is required.
if(NOT DEFINED IDF_TARGET)
  set(IDF_TARGET esp32p4 CACHE STRING "SDK target")
endif()
set(MOTO_SDK_DIR "${CMAKE_CURRENT_LIST_DIR}/../../vendor/moto-sdk" CACHE PATH "Imported binary SDK")
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/check_sdk.cmake")
set(SDKCONFIG_DEFAULTS "${MOTO_SDK_DIR}/components/moto_prebuilt/sdkconfig.defaults;${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
list(APPEND EXTRA_COMPONENT_DIRS "${MOTO_SDK_DIR}/components/moto_prebuilt")
option(MOTO_DEMO_MJPEG "Enable the real MJPEG decoder integration path" OFF)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(moto_customer)
